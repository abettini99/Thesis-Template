% Header cells: LaTeX gives their Scope through attribute classes (/C /TH-col ...); PAC reads only
% direct attributes, so also write the Scope on the cell itself (/A), keeping the class.
\cs_new_eq:NN \__thesis_struct_write_obj:n \__tag_struct_write_obj:n
\cs_set_protected:Npn \__tag_struct_write_obj:n #1
  {
    \prop_get:cnNT { g__tag_struct_#1_prop } { C } \l_tmpa_tl
      {
        \tl_clear:N \l_tmpb_tl
        \tl_if_in:NnT \l_tmpa_tl { /TH-col } { \tl_set:Nn \l_tmpb_tl { /Column } }
        \tl_if_in:NnT \l_tmpa_tl { /TH-row } { \tl_set:Nn \l_tmpb_tl { /Row } }
        \tl_if_in:NnT \l_tmpa_tl { /TH-both } { \tl_set:Nn \l_tmpb_tl { /Both } }
        \tl_if_empty:NF \l_tmpb_tl
          {
            \prop_get:cnNTF { g__tag_struct_#1_prop } { A } \l_tmpa_tl
              {
                \tl_remove_all:Nn \l_tmpa_tl { [ }
                \tl_remove_all:Nn \l_tmpa_tl { ] }
                \prop_gput:cne { g__tag_struct_#1_prop } { A }
                  { [ \l_tmpa_tl \c_space_tl << /O /Table /Scope \l_tmpb_tl >> ] }
              }
              {
                \prop_gput:cne { g__tag_struct_#1_prop } { A }
                  { << /O /Table /Scope \l_tmpb_tl >> }
              }
          }
      }
    \__thesis_struct_write_obj:n {#1}
  }
\ExplSyntaxOff
