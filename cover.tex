%% COVER
%% Back, spine and front on one sheet, with bleed, as printers ask for a paperback cover.
%% Compile it on its own: in Overleaf, open this file and choose Menu > Main document > cover.tex
%% (switch back to main.tex after), or locally run: latexmk cover.tex
%% Title and name come from _thesisinfo.tex.
%%
%% Before sending it to the printer:
%%   1. Set the page size below to the one the thesis uses.
%%   2. Set the page count to the number of pages of the FINAL thesis PDF (main.pdf, print-preparation edition),
%%      including the blank pages.
%%   3. Ask the printer for the paper thickness (or the spine width) and set it below.
%%   4. Ask the printer how much bleed they want on the cover (often 3 to 5 mm; more for hardcovers).
%%   5. Turn the guides off (\coverguidesfalse) and send cover.pdf.
%% Many printers also have a cover template or calculator; check the result against it.
%% =====================================================================
%% Title and name, and the archival PDF (PDF/A-4f; set up before \documentclass)

%% Book page size (trimmed), the same as the class option in main.tex:
%%   demy 138 x 216   a5 148 x 210   royal 156 x 234   tudelft 170 x 240   (mm)
\newcommand{\coverpagewidth}{138}   % mm
\newcommand{\coverpageheight}{216}  % mm

%% Spine width = number of pages / 2 x thickness of one sheet (leaf) + extra for the cover board/glue.
\newcommand{\coverpagecount}{200}         % pages of the final thesis PDF, including blank pages
\newcommand{\coverpaperthickness}{0.110}  % mm per sheet; about 0.10 mm for 80 g/m2, 0.11 mm for 90 g/m2,
                                          % 0.12-0.13 mm for 100 g/m2 uncoated paper. Ask the printer!
\newcommand{\coverspineextra}{0.5}        % mm, for the cover itself; the printer will tell you (0 if unsure)
%% Or set the spine width the printer gives you directly, and ignore the three lines above:
% \newcommand{\coverspinefixed}{12.5}     % mm

\newcommand{\coverbleed}{3}  % mm on all four sides
\newcommand{\coversafe}{5}   % mm: keep text at least this far inside the trim and away from the spine folds

%% Show trim lines, spine folds and the safe area (for checking only; turn off for the printer).
\newif\ifcoverguides
\coverguidestrue   % \coverguidesfalse for the final file

%% Spine text direction: -90 reads from top to bottom (English convention, and most Dutch books
%% today), 90 from bottom to top (traditional continental convention).
\newcommand{\coverspinerotation}{-90}

\definecolor{coverback}{HTML}{CFE6E6}   % pale teal

%% Front: the title in capitals, with your own line breaks (\\); the subtitle and a small line under the
%% name. The title is made smaller automatically when its longest line is too wide.
\newcommand{\coverleadin}{Title of the}  % a short lead-in line above the title, smaller: e.g. Adjoint-based; {} for none
\newcommand{\covertitle}{Dissertation}   % the rest of the title, large: e.g. Inflow\\ Modelling (or \thesistitle)
\newcommand{\coversubtitle}{}            % a line under the title, smaller: e.g. \thesissubtitle; {} for none
\newcommand{\coversmallprint}{Doctoral thesis\quad·\quad TU Delft}
\newcommand{\coverspinemark}{TU Delft}   % small, at the foot of the spine: TU Delft, the year, ...; {} for none
%% Or a logo at the foot of the spine instead, turned with the spine text: an image file, best a vector PDF
%% in black on a transparent background. Its height is measured across the spine. Leave empty to print
%% \coverspinemark instead.
\newcommand{\coverspinelogo}{}        % e.g. Images/_title_images/logo_black.pdf, the TU Delft logo of the title page
\newcommand{\coverspinelogoheight}{6.5} % mm (at most 60 % of the spine width)

%% The drawing in the middle of the front: TikZ code (lengths in mm, centred on (0,0), about 40 mm across),
%% or an image: \renewcommand{\covermotif}{\node {\includegraphics[width=45mm]{Images/motif.pdf}};}
%% The default is a patch of a triangular mesh.
\newcommand{\covermotif}{%
    \begin{scope}[x={(9mm,0)}, y={(4.5mm,7.794mm)}, line width=0.5pt, line join=round]
        \foreach \i in {-2,...,2} \foreach \j in {-2,...,2} {
            \pgfmathtruncatemacro\inside{abs(\i+\j) <= 2}
            \ifnum\inside=1
                \pgfmathtruncatemacro\right{\i < 2 && abs(\i+\j+1) <= 2}
                \pgfmathtruncatemacro\up{\j < 2 && abs(\i+\j+1) <= 2}
                \pgfmathtruncatemacro\down{\i < 2 && \j > -2}
                \ifnum\right=1 \draw (\i,\j) -- (\i+1,\j); \fi
                \ifnum\up=1 \draw (\i,\j) -- (\i,\j+1); \fi
                \ifnum\down=1 \draw (\i,\j) -- (\i+1,\j-1); \fi
                \fill (\i,\j) circle (0.9mm);
            \fi}
    \end{scope}}

%% ISBN barcode on the back: an image file from the printer or an online barcode generator (vector PDF
%% or EPS is best). Leave empty to keep a white box, where the printer places the barcode.
\newcommand{\coverbarcode}{}   % e.g. Images/barcode.pdf

%% Text on the back: the summary (paragraphs separated by \par\medskip), then a small spaced heading,
%% ABOUT THE AUTHOR, and a short biography.
\newcommand{\coversummary}{%
    A short summary of the thesis for the back cover, in two or three paragraphs: what the problem is,
    what was done and what was found.\par\medskip
    Write it for an interested reader who is not a specialist.}
\newcommand{\coverbio}{%
    A short biography of the author.}

\ExplSyntaxOn
\cs_if_exist:NTF \coverspinefixed
    { \edef\coverspine{\fp_eval:n { round(\coverspinefixed, 2) }} }
    { \edef\coverspine{\fp_eval:n { round(\coverpagecount / 2 * \coverpaperthickness + \coverspineextra, 2) }} }
\edef\coversheetwidth {\fp_eval:n { 2*\coverbleed + 2*\coverpagewidth + \coverspine }}
\edef\coversheetheight{\fp_eval:n { 2*\coverbleed + \coverpageheight }}
%% x positions of the trim and fold lines, measured from the left edge of the sheet
\edef\coverbackleft  {\fp_eval:n { \coverbleed }}
\edef\coverspineleft {\fp_eval:n { \coverbleed + \coverpagewidth }}
\edef\coverspineright{\fp_eval:n { \coverbleed + \coverpagewidth + \coverspine }}
\edef\coverfrontright{\fp_eval:n { \coverbleed + 2*\coverpagewidth + \coverspine }}
\edef\coverbottom    {\fp_eval:n { \coverbleed }}
\edef\covertop       {\fp_eval:n { \coverbleed + \coverpageheight }}
\ExplSyntaxOff
\typeout{}\typeout{Cover: spine width \coverspine\space mm, sheet \coversheetwidth\space x \coversheetheight\space mm (including \coverbleed\space mm bleed)}\typeout{}

%% Cover font: Josefin Sans (SIL Open Font License, in _options/josefin), a variable font: regular at weight

\newcommand{\covertrack}[1]{\addfontfeatures{LetterSpace=#1}} % tracking, in 1/100 em: negative is tighter

%% Back-cover text block: the summary in the book's serif at 10.5 on 13.5 pt, then a small spaced
%% heading in the cover font and the biography.
\newcommand{\backcovertext}[2]{%
    \fontsize{10.5}{13.5}\selectfont
    #1\par\bigskip
    {\coverfont\fontsize{8}{10}\selectfont\covertrack{20}ABOUT THE AUTHOR}\par\smallskip
    #2}

\pagestyle{empty}

\setlength{\parskip}{0.6\baselineskip}

\newsavebox{\covertitlebox}
\newsavebox{\coverspinebox}
\newlength{\coverspinemax}
\newcommand{\spinetitlebox}[1]{% the spine title at #1 pt, scaled down to 65 % of the page height if longer
    \sbox{\coverspinebox}{\coverfont\fontsize{#1}{#1}\selectfont\covertrack{-3}% tight, like the front title
        \ifx\coverleadin\empty\else\coverspineleadin\ \fi\textbf{\coverspinetitle}}%
    \setlength{\coverspinemax}{0.65\dimexpr\coverpageheight mm\relax}%
    \ifdim\wd\coverspinebox>\coverspinemax \resizebox{\coverspinemax}{!}{\usebox{\coverspinebox}}\else\usebox{\coverspinebox}\fi}
\begin{document}
\ExplSyntaxOn % the title in capitals, its \\ kept
\tl_gset:Ne \g_tmpa_tl { \text_uppercase:n { \covertitle } }
\tl_gset:Ne \g_tmpb_tl { \text_uppercase:n { \covertitle } } % the spine title in capitals, on one line
\tl_greplace_all:Nnn \g_tmpb_tl { \\ } { \unskip \space \ignorespaces }
\cs_gset:Npe \coverspinetitle { \exp_not:V \g_tmpb_tl }
\cs_gset:Npe \coverspineleadin { \text_uppercase:n { \coverleadin } }
\sbox{\covertitlebox}{\coverfont\bfseries\covertrack{-8}\fontsize{56}{54}\selectfont
    \begin{tabular}{@{}c@{}}\g_tmpa_tl\end{tabular}}
\ExplSyntaxOff
\begin{tikzpicture}[remember picture, overlay, x=1mm, y=1mm, shift={(current page.south west)}]
    \pgfmathsetmacro{\frontcentre}{(\coverspineright + \coverfrontright)/2}
    \pgfmathsetmacro{\backcentre}{(\coverbackleft + \coverspineleft)/2}
    \pgfmathsetmacro{\covertextwidth}{\coverpagewidth - 2*\coversafe - 6}

    %% One colour over the whole sheet, bleed included
    \fill[coverback] (0, 0) rectangle (\coversheetwidth, \coversheetheight);

    %% Front: title and subtitle at the top, the drawing in the middle, name and small print at the bottom
    \coordinate (titletop) at (\frontcentre, \covertop - 0.12*\coverpageheight);
    \if\relax\detokenize\expandafter{\coverleadin}\relax\else
        \node[anchor=north, align=center, text width=\covertextwidth mm, text=covertext, inner sep=0,
              font={\coverfont\covertrack{-4}\fontsize{27}{30}\selectfont}]
            (leadin) at (titletop) {\MakeUppercase{\coverleadin}};
        \coordinate (titletop) at ([yshift=-1.5mm]leadin.south);
    \fi
    \node[anchor=north, inner sep=0, text=covertext] (title) at (titletop)
        {\resizebox{\ifdim\wd\covertitlebox>\covertextwidth mm \covertextwidth mm\else\wd\covertitlebox\fi}{!}{\usebox\covertitlebox}};
    \if\relax\detokenize\expandafter{\coversubtitle}\relax\else
        \node[anchor=north, align=center, text width=\covertextwidth mm, text=covertext, inner sep=0,
              font={\coverfont\covertrack{-2}\fontsize{15}{19}\selectfont}]
            at ([yshift=-7mm]title.south) {\MakeUppercase{\coversubtitle}};
    \fi
    \begin{scope}[shift={(\frontcentre, \coverbottom + 0.46*\coverpageheight)}, covermotifcolour, x=1mm, y=1mm]
        \covermotif
    \end{scope}
    \node[anchor=south, align=center, text width=\covertextwidth mm, text=covertext, inner sep=0,
          font={\coverfont\covertrack{-6}\fontsize{24}{28}\selectfont}]
        at (\frontcentre, \coverbottom + 24) {\thesisauthorfirstnames\ \thesisauthorlastname};
    \node[anchor=south, text=covertext!75, inner sep=0, font={\coverfont\covertrack{20}\fontsize{6.5}{8}\selectfont}]
        at (\frontcentre, \coverbottom + 14) {\MakeUppercase{\coversmallprint}};

    %% Spine, if it is wide enough for text: the name near the top, the title in the middle (the lead-in
    %% at weight 350 and the title at 500, as on the front), a small mark
    %% near the bottom (\coverspinemark)
    \pgfmathparse{\coverspine >= 6 ? 1 : 0}
    \ifnum\pgfmathresult=1
        \pgfmathsetmacro{\spinefont}{min(10, 0.55*\coverspine/0.3528)}   % pt, at most 10 pt
        \pgfmathsetmacro{\spinemarkfont}{0.7*\spinefont}
        \pgfmathsetmacro{\spinetitlefont}{min(22, 0.48*\coverspine/(0.702*0.3528))} % pt: capitals 48 % of the spine width, at most 22 pt
        \pgfmathsetmacro{\spinex}{(\coverspineleft + \coverspineright)/2}
        \ifnum\coverspinerotation<0 \def\spinestart{west}\def\spineend{east}\else\def\spinestart{east}\def\spineend{west}\fi
        \begin{scope}[every node/.style={rotate=\coverspinerotation, text=covertext, inner sep=0}]
            \node[anchor=\spinestart] at (\spinex, \covertop - \coversafe - 8)
                {\coverfont\bfseries\fontsize{\spinefont}{\spinefont}\selectfont \MakeUppercase{\thesisauthorlastname}};
            \node[anchor=center] at (\spinex, \coverbottom + 0.5*\coverpageheight)
                {\spinetitlebox{\spinetitlefont}}; % a long title is made smaller, to 65 % of the height
            \if\relax\detokenize\expandafter{\coverspinelogo}\relax
                \node[anchor=\spineend] at (\spinex, \coverbottom + \coversafe + 8)
                    {\coverfont\covertrack{20}\fontsize{\spinemarkfont}{\spinefont}\selectfont \MakeUppercase{\coverspinemark}};
            \else
                \pgfmathsetmacro{\spinelogoheight}{min(\coverspinelogoheight, 0.6*\coverspine)}
                \node[anchor=\spineend] at (\spinex, \coverbottom + \coversafe + 8)
                    {\includegraphics[height=\spinelogoheight mm, trim=3.9bp 3.5bp 10.5bp 3.6bp, clip]{\coverspinelogo}}; % trimmed to the logo itself (TU Delft file)
            \fi
        \end{scope}
    \else
        \typeout{Cover: the spine is narrower than 6 mm, so no text is printed on it.}
    \fi

    %% Back: the summary in the text font of the book at 10.5 pt, its top level with the front title, at a
    %% comfortable line length, then the biography; space for the ISBN barcode at the bottom
    \pgfmathsetmacro{\blurbwidth}{min(\covertextwidth, 100)}
    \node[anchor=north, text width=\blurbwidth mm, text=covertext, inner sep=0,
          font=\rmfamily]
        at (\backcentre, \covertop - 0.12*\coverpageheight)
        {\setlength{\parindent}{0pt}\backcovertext{\coversummary}{\coverbio}};
    \begin{scope}[shift={(\coverspineleft - \coversafe - 40, \coverbottom + \coversafe)}]
        \fill[white] (0, 0) rectangle (40, 22);
        \if\relax\detokenize\expandafter{\coverbarcode}\relax
            \ifcoverguides\node[text=black!50, font=\sffamily\scriptsize, align=center] at (20, 11) {ISBN \thesisisbn\\barcode area};\fi
        \else
            \node[inner sep=0] at (20, 11) {\includegraphics[width=36mm, height=18mm, keepaspectratio]{\coverbarcode}};
        \fi
    \end{scope}

    %% Guides (not for the printer)
    \ifcoverguides
        \begin{scope}[line width=0.4pt]
            \draw[red] (\coverbackleft, \coverbottom) rectangle (\coverfrontright, \covertop);        % trim
            \draw[red, dashed] (\coverspineleft, 0) -- (\coverspineleft, \coversheetheight)          % spine folds
                               (\coverspineright, 0) -- (\coverspineright, \coversheetheight);
            \draw[blue, densely dotted]                                                                % safe area
                (\coverbackleft + \coversafe, \coverbottom + \coversafe) rectangle (\coverspineleft - \coversafe, \covertop - \coversafe)
                (\coverspineright + \coversafe, \coverbottom + \coversafe) rectangle (\coverfrontright - \coversafe, \covertop - \coversafe);
            \node[red, font=\sffamily\tiny, anchor=north, fill=white, inner sep=1pt]
                at ({(\coverspineleft + \coverspineright)/2}, \covertop - 2)
                {spine \coverspine\,mm};
            \node[red, font=\sffamily\tiny, anchor=south west, fill=white, inner sep=1pt]
                at (\coverbackleft + 1, \covertop + 0.3)
                {back \quad sheet \coversheetwidth\,×\,\coversheetheight\,mm incl.\ \coverbleed\,mm bleed \quad guides on: set \textbackslash coverguidesfalse for the printer};
        \end{scope}
    \fi
\end{tikzpicture}
