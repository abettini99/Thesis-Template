%% =====================================================================
%% COVER
%% Back, spine and front on one sheet, with bleed, as printers ask for a paperback cover.
%% Compile it on its own: in Overleaf, open this file and choose Menu > Main document > cover.tex
%% (switch back to main.tex after), or locally run: latexmk cover.tex
%% The texts, colours and drawing are set in _coversettings.tex (the digital edition of the thesis uses
%% them too, for its first and last pages); title and name come from _thesisinfo.tex.
%%
%% Before sending it to the printer (in _coversettings.tex):
%%   1. Set the page size to the one the thesis uses.
%%   2. Set the page count to the number of pages of the FINAL thesis PDF (main.pdf, print-preparation edition),
%%      including the blank pages.
%%   3. Ask the printer for the paper thickness (or the spine width) and set it.
%%   4. Ask the printer how much bleed they want on the cover (often 3 to 5 mm; more for hardcovers).
%%   5. Turn the guides off (\coverguidesfalse) and send cover.pdf.
%% Many printers also have a cover template or calculator; check the result against it.
%% =====================================================================
%% For the printer: PDF/X-6 or PDF/X-4 with the printer's colour profile (set in _thesisinfo.tex, as for the
%% thesis), set up before \documentclass; title and name from _thesisinfo.tex
\def\thesisforprinter{}
\input{_options/documentmetadata}
\def\thesispdfsuffix{ (cover)}
\documentclass{article}
\usepackage{fix-cm} % maths fonts at any size (the large title)
\AddToHook{begindocument/end}{% (as fix-cm does; again, as LaTeX sets it at the start of the document)
    \DeclareFontShape{OMX}{cmex}{m}{n}{<-7.5>cmex7<7.5-8.5>cmex8<8.5-9.5>cmex9<9.5->cmex10}{}}
\input{_options/pdfa} % title, author and language in the metadata

\usepackage{xcolor}
\input{_options/formatColor}
\input{_coversettings}

%% ---------------------------------------------------------------------
%% Geometry (no changes needed below this line)
%% ---------------------------------------------------------------------
\ExplSyntaxOn
\cs_if_exist:NTF \coverspinefixed
    { \edef\coverspine{\fp_eval:n { round(\coverspinefixed, 2) }} }
    { \edef\coverspine{\fp_eval:n { round(\coverpagecount / 2 * \coverpaperthickness + \coverspineextra, 2) }} }
\edef\coversheetwidth {\fp_eval:n { 2*\coverbleed + 2*\coverpagewidth + \coverspine }}
\edef\coversheetheight{\fp_eval:n { 2*\coverbleed + \coverpageheight }}
%% x positions of the trim and fold lines, measured from the left edge of the sheet
\edef\coverbackleft  {\fp_eval:n { \coverbleed }}
\edef\coverspineleft {\fp_eval:n { \coverbleed + \coverpagewidth }}
\edef\coverspineright{\fp_eval:n { \coverbleed + \coverpagewidth + \coverspine }}
\edef\coverfrontright{\fp_eval:n { \coverbleed + 2*\coverpagewidth + \coverspine }}
\edef\coverbottom    {\fp_eval:n { \coverbleed }}
\edef\covertop       {\fp_eval:n { \coverbleed + \coverpageheight }}
\ExplSyntaxOff
\typeout{}\typeout{Cover: spine width \coverspine\space mm, sheet \coversheetwidth\space x \coversheetheight\space mm (including \coverbleed\space mm bleed)}\typeout{}

\usepackage[paperwidth=\coversheetwidth mm, paperheight=\coversheetheight mm, margin=0pt]{geometry}
%% For the printer: the trimmed cover inside the bleed (page boxes), RGB as sRGB (_options/printpdf.tex)
\input{_options/printpdf}
\thesisprintboxes{\coverbleed mm}{\coverbleed mm}{\dimexpr\coversheetwidth mm-2\dimexpr\coverbleed mm\relax\relax}{\coverpageheight mm}{\coverbleed mm}

\begin{document}
\coverprepare
\begin{tikzpicture}[remember picture, overlay, x=1mm, y=1mm, shift={(current page.south west)}]
    %% One colour over the whole sheet, bleed included
    \fill[coverback] (0, 0) rectangle (\coversheetwidth, \coversheetheight);

    %% Front and back (_options/coverdesign.tex)
    \coverfront{\coverspineright}
    \coverback{\coverbackleft}{1}

    %% Spine, if it is wide enough for text: the name near the top, the title in the middle (the lead-in
    %% at weight 350 and the title at 500, as on the front), a small mark
    %% near the bottom (\coverspinemark)
    \pgfmathparse{\coverspine >= 6 ? 1 : 0}
    \ifnum\pgfmathresult=1
        \pgfmathsetmacro{\spinefont}{min(10, 0.55*\coverspine/0.3528)}   % pt, at most 10 pt
        \pgfmathsetmacro{\spinemarkfont}{0.7*\spinefont}
        \pgfmathsetmacro{\spinetitlefont}{min(22, 0.48*\coverspine/(0.702*0.3528))} % pt: capitals 48 % of the spine width, at most 22 pt
        \pgfmathsetmacro{\spinex}{(\coverspineleft + \coverspineright)/2}
        \ifnum\coverspinerotation<0 \def\spinestart{west}\def\spineend{east}\else\def\spinestart{east}\def\spineend{west}\fi
        \begin{scope}[every node/.style={rotate=\coverspinerotation, text=covertext, inner sep=0}]
            \node[anchor=\spinestart] at (\spinex, \covertop - \coversafe - 8)
                {\coverfont\bfseries\fontsize{\spinefont}{\spinefont}\selectfont \MakeUppercase{\thesisauthorlastname}};
            \node[anchor=center] at (\spinex, \coverbottom + 0.5*\coverpageheight)
                {\spinetitlebox{\spinetitlefont}}; % a long title is made smaller, to 65 % of the height
            \if\relax\detokenize\expandafter{\coverspinelogo}\relax
                \node[anchor=\spineend] at (\spinex, \coverbottom + \coversafe + 8)
                    {\coverfont\covertrack{20}\fontsize{\spinemarkfont}{\spinefont}\selectfont \MakeUppercase{\coverspinemark}};
            \else
                \pgfmathsetmacro{\spinelogoheight}{min(\coverspinelogoheight, 0.6*\coverspine)}
                \node[anchor=\spineend] at (\spinex, \coverbottom + \coversafe + 8)
                    {\includegraphics[height=\spinelogoheight mm, trim=3.9bp 3.5bp 10.5bp 3.6bp, clip]{\coverspinelogo}}; % trimmed to the logo itself (TU Delft file)
            \fi
        \end{scope}
    \else
        \typeout{Cover: the spine is narrower than 6 mm, so no text is printed on it.}
    \fi

    %% Guides (not for the printer)
    \ifcoverguides
        \begin{scope}[line width=0.4pt]
            \draw[red] (\coverbackleft, \coverbottom) rectangle (\coverfrontright, \covertop);        % trim
            \draw[red, dashed] (\coverspineleft, 0) -- (\coverspineleft, \coversheetheight)          % spine folds
                               (\coverspineright, 0) -- (\coverspineright, \coversheetheight);
            \draw[blue, densely dotted]                                                                % safe area
                (\coverbackleft + \coversafe, \coverbottom + \coversafe) rectangle (\coverspineleft - \coversafe, \covertop - \coversafe)
                (\coverspineright + \coversafe, \coverbottom + \coversafe) rectangle (\coverfrontright - \coversafe, \covertop - \coversafe);
            \node[red, font=\sffamily\tiny, anchor=north, fill=white, inner sep=1pt]
                at ({(\coverspineleft + \coverspineright)/2}, \covertop - 2)
                {spine \coverspine\,mm};
            \node[red, font=\sffamily\tiny, anchor=south west, fill=white, inner sep=1pt]
                at (\coverbackleft + 1, \covertop + 0.3)
                {back \quad sheet \coversheetwidth\,×\,\coversheetheight\,mm incl.\ \coverbleed\,mm bleed \quad guides on: set \textbackslash coverguidesfalse for the printer};
        \end{scope}
    \fi
\end{tikzpicture}
\end{document}
