%% =====================================================================
%% NOMENCLATURE: define acronyms and symbols here (this file is loaded in the preamble).
%% Use them in the text with \gls{label}: acronyms are written out in full the first time,
%% symbols are printed as defined. The lists are printed in Chapters/00Nomenclature.tex.
%%
%%   \nomacronym[sort]{label}{SHORT}{long form}
%%   \nomdimless[sort]{label}{symbol}{description}{definition}
%%   \nomlatin[sort]{label}{symbol}{description}{dimensions}{units}
%%   \nomgreek[sort]{label}{symbol}{description}{dimensions}{units}
%%   \nomsub[sort]{label}{symbol}{description}
%%   \nomsuper[sort]{label}{symbol}{description}
%%   \nomnotation[sort]{label}{symbol}{description}
%%
%% Symbols, dimensions and definitions are written as maths, without $...$.
%% Write -- for a dimensionless quantity. Dimensions are only listed with the style nom-dimensions
%% (Chapters/00Nomenclature.tex); fill them in anyway, so that switching costs nothing. Labels must be unique across all lists.
%% Latin symbols are sorted by their sort key (the label if none is given); Greek symbols,
%% sub- and superscripts and the notation appear in the order they are defined here.
%% Give every symbol one meaning: where two quantities share a letter, tell them apart with a
%% subscript or another letter.
%% =====================================================================
