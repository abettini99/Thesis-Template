%% =====================================================================
%% The PDF standard of each edition, set before \documentclass (as LaTeX requires) from the edition
%% named in the \documentclass line that follows:
%%   digital, proof: archival, PDF/A-4f; in the final version (version=final) also accessible,
%%       PDF/UA-2, tagged for screen readers, with MathML (drafts are not tagged: faster compiles);
%%   preview: a plain PDF (2.0), not tagged and without attachments: the book as printed, for looking
%%       at, and not a PDF/X file, so that it cannot be sent to the printer by mistake (it has no bleed);
%%   printer: PDF/X-6 or PDF/X-4 (\thesisprintstandard in settings/thesisinfo.tex),
%%       for the printer's colour profile (\thesisprintprofile) or FOGRA39, not tagged.
%% Read by main.tex, propositions.tex and cover.tex (always for the printer: \thesisforprinter) on their
%% first line. It reads settings/thesisinfo.tex (the class does not read it again).
%% =====================================================================
\input{settings/thesisinfo}
\expandafter\ifx\csname ifthesistagging\endcsname\relax \expandafter\newif\csname ifthesistagging\endcsname \csname thesistaggingtrue\endcsname\fi
\providecommand{\thesisyear}{\the\year}
\providecommand{\thesispdftitle}{\thesistitle\if\relax\detokenize\expandafter{\thesissubtitle}\relax\else: \thesissubtitle\fi} % title: subtitle
\providecommand{\thesisprintstandard}{X-6}
\providecommand{\thesisprintprofile}{}
\providecommand{\thesisbleed}{2mm}

%% The printer's colour profile: its colour space and description, read from the file (ICC header and
%% "desc" tag), for the PDF's output intent. Without one: FOGRA39 (offset print on coated paper), which
%% comes with TeX Live.
\directlua{dofile(kpse.find_file("thesisprint.lua", "lua") or "template/thesisprint.lua")}

\ExplSyntaxOn
\cs_new_protected:Npn \__thesis_metadata_print:
  {
    \tl_if_blank:VTF \thesisprintprofile
      { \tl_set:Nn \l__thesis_profile_tl { FOGRA39L_coated.icc } }
      {
        \tl_set:Ne \l__thesis_profile_tl { \tl_trim_spaces:V \thesisprintprofile }
        \file_if_exist:VF \l__thesis_profile_tl
          { \msg_error:nnV { thesis } { profile-missing } \l__thesis_profile_tl }
        % describe it for the output intent, as LaTeX does for the profiles it knows
        \directlua { thesis.iccinfo("\l__thesis_profile_tl") }
        \tl_if_empty:NTF \thesisprofileclass
          { \msg_error:nnV { thesis } { profile-unreadable } \l__thesis_profile_tl }
          {
            \str_if_eq:VnF \thesisprofileclass { prtr }
              { \msg_error:nnV { thesis } { profile-class } \l__thesis_profile_tl }
            \tl_if_empty:NT \thesisprofiledesc { \tl_set_eq:NN \thesisprofiledesc \l__thesis_profile_tl }
            \prop_const_from_keyval:ce { c__pdfmeta_colorprofile_ \l__thesis_profile_tl }
              {
                OutputConditionIdentifier = { \exp_not:V \thesisprofiledesc } ,
                Info = { \exp_not:V \thesisprofiledesc } ,
                N = \str_case:Vn \thesisprofilespace { { CMYK } { 4 } { RGB } { 3 } { GRAY } { 1 } }
              }
          }
      }
    \str_case:enF { \str_uppercase:f { \tl_trim_spaces:V \thesisprintstandard } }
      {
        { X-6 } { \tl_set:Nn \l__thesis_pdfx_tl { X-6 } \tl_set:Nn \l__thesis_pdfversion_tl { 2.0 } }
        { X-4 } { \tl_set:Nn \l__thesis_pdfx_tl { X-4 } \tl_set:Nn \l__thesis_pdfversion_tl { 1.6 } }
      }
      {
        \msg_error:nne { thesis } { print-standard } { \thesisprintstandard }
        \tl_set:Nn \l__thesis_pdfx_tl { X-6 } \tl_set:Nn \l__thesis_pdfversion_tl { 2.0 }
      }
    \cs_gset:Npe \thesispdfx { \l__thesis_pdfx_tl } % (the class and pdfa.tex test it)
    \use:e
      {
        \exp_not:N \DocumentMetadata
          {
            pdfversion = \l__thesis_pdfversion_tl ,
            pdfstandard = \l__thesis_pdfx_tl ,
            colorprofiles = { X = \l__thesis_profile_tl }
          }
      }
  }
\cs_new_protected:Npn \__thesis_metadata_archival:
  {
    \ifthesistagging
      \DocumentMetadata
        {
          pdfversion = 2.0 , pdfstandard = { ua-2 , a-4f } , tagging = on ,
          tagging-setup = { math/setup = { mathml-SE , mathml-AF } }
        }
    \else
      \DocumentMetadata { pdfversion = 2.0 , pdfstandard = a-4f }
    \fi
  }
\cs_new_protected:Npn \__thesis_metadata_plain:
  {
    \cs_gset:Npn \thesisplainpdf { } % (attachments.tex tests it)
    \DocumentMetadata { pdfversion = 2.0 }
  }
\tl_new:N \l__thesis_profile_tl
\tl_new:N \l__thesis_pdfx_tl
\tl_new:N \l__thesis_pdfversion_tl
\cs_generate_variant:Nn \file_if_exist:nF { V }
\cs_generate_variant:Nn \prop_const_from_keyval:Nn { ce }
\cs_generate_variant:Nn \str_case:nn { V }
\msg_new:nnn { thesis } { profile-missing }
  { Colour~profile~`#1'~(\token_to_str:N \thesisprintprofile\ in~settings/thesisinfo.tex)~not~found. }
\msg_new:nnn { thesis } { profile-unreadable }
  { `#1'~is~not~an~ICC~colour~profile~(\token_to_str:N \thesisprintprofile\ in~settings/thesisinfo.tex). }
\msg_new:nnn { thesis } { profile-class }
  { `#1'~is~not~a~printer~profile;~PDF/X~needs~the~profile~of~the~printing~press~(ask~your~printer). }
\msg_new:nnn { thesis } { print-standard }
  { \token_to_str:N \thesisprintstandard\ is~`#1';~use~X-6~or~X-4.~Using~X-6. }

%% The class options that matter here: edition=printer (PDF/X) or preview (plain), and version (tagged
%% only when final)
\cs_new_protected:Npn \__thesis_dm_key:n #1 { }
\cs_new_protected:Npn \__thesis_dm_keyval:nn #1#2
  {
    \str_if_eq:nnT {#1} { edition }
      {
        \str_if_eq:nnTF {#2} { printer } { \bool_set_true:N \l_tmpa_bool } { \bool_set_false:N \l_tmpa_bool }
        \str_if_eq:nnTF {#2} { preview } { \bool_set_true:N \l_tmpb_bool } { \bool_set_false:N \l_tmpb_bool }
      }
    \str_if_eq:nnT {#1} { version }
      { \str_if_eq:nnTF {#2} { final } { \legacy_if_gset_true:n { thesistagging } } { \legacy_if_gset_false:n { thesistagging } } }
  }
%% \documentclass, once: look at the edition, set the standard, then load the class as usual
\NewCommandCopy \__thesis_documentclass: \documentclass
\RenewDocumentCommand \documentclass { o m }
  {
    \bool_set_false:N \l_tmpa_bool
    \bool_set_false:N \l_tmpb_bool
    \legacy_if_gset_false:n { thesistagging } % (version=draft unless the options say otherwise)
    \IfValueT {#1} { \keyval_parse:NNn \__thesis_dm_key:n \__thesis_dm_keyval:nn {#1} }
    \cs_if_exist:NT \thesisforprinter { \bool_set_true:N \l_tmpa_bool } % (cover.tex: always for the printer)
    \bool_if:NTF \l_tmpa_bool
      { \__thesis_metadata_print: }
      { \bool_if:NTF \l_tmpb_bool { \__thesis_metadata_plain: } { \__thesis_metadata_archival: } }
    \RenewCommandCopy \documentclass \__thesis_documentclass:
    \IfValueTF {#1} { \documentclass [#1] {#2} } { \documentclass {#2} }
  }
\ExplSyntaxOff
