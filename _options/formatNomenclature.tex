\newlength\descriptionwidth
\newlength\dimensionswidth
\newlength\unitswidth
\newlength\definitionwidth
\newlength\symbolwidth
%% Column widths (book page, used in both print and report mode)
\setlength\symbolwidth{4.5em}
\setlength\dimensionswidth{7em}
\setlength\unitswidth{4.5em}
%% Description and definition columns take whatever width is left, so the nomenclature fits every page size.
\setlength\descriptionwidth{\dimexpr\textwidth-\symbolwidth-\dimensionswidth-\unitswidth-1em\relax}
\setlength\definitionwidth{\descriptionwidth}

\newcommand{\nomdescr}[1]{\parbox[t]{\descriptionwidth}{\RaggedRight #1}} % the cm box changes the size of the description box
\newcommand{\DimensUnits}[2]{\hfill\makebox[\dimensionswidth]{#1\hfill}%
\makebox[\unitswidth]{#2\hfill}\ignorespaces} % first number related to 'dimensions' box size, second number related to 'units' box size
\newcommand{\DefinitionCol}[1]{\hfill\parbox[t]{\definitionwidth}{#1}\ignorespaces} % number related to 'definition' box size
