%% =====================================================================
%% Nomenclature with glossaries-extra.
%% Entries are defined in _nomenclature.tex (preamble) and printed in Chapters/00Nomenclature.tex.
%% In the text:  \gls{bem}   -> "blade element momentum (BEM)" the first time, "BEM" after that
%%               \gls{rho}   -> the symbol, in text or maths
%% No external tool is needed: sorting is done by LaTeX itself (\makenoidxglossaries).
%% =====================================================================
\makeatletter
%% glossaries loads datatool, which (v3, TeX Live 2025 and later) warns that it has no Dutch localisation
%% because babel loads Dutch for the summary. That is harmless here (nothing is sorted in Dutch), so turn the
%% warning off. Only for datatool v3, recognised by its rollback file; older versions don't know the option.
\IfFileExists{datatool-base-2019-09-27.sty}{%
    \PassOptionsToPackage{lang-warn=false}{datatool-base}%
    \PassOptionsToPackage{nowarn}{tracklang}%
}{}
\usepackage[nomain, nopostdot, nonumberlist, nogroupskip]{glossaries-extra}
\makenoidxglossaries
%% Store entry fields exactly as written (units such as \unit{\metre} only work when typeset).
\glsnoexpandfields

%% One list per kind of entry.
\newglossary*{acronyms}{Acronyms}
\newglossary*{dimless}{Dimensionless numbers}
\newglossary*{latin}{Latin symbols}
\newglossary*{greek}{Greek symbols}
\newglossary*{super}{Superscripts}
\newglossary*{sub}{Subscripts}

%% Acronyms: long form with the short form in brackets on first use, short form afterwards.
\setabbreviationstyle[acronym]{long-short}

%% Extra columns for symbols.
\glsaddstoragekey{dimensions}{}{\glsentrydimensions}
\glsaddstoragekey{units}{}{\glsentryunits}
\glsaddstoragekey{definition}{}{\glsentrydefinition}

%% ---------------------------------------------------------------------
%% Commands for defining entries (use them in _nomenclature.tex).
%% The optional first argument is a sort key. By default, acronyms are sorted by their short
%% form and Latin symbols and dimensionless numbers by their label; Greek symbols and sub- and
%% superscripts keep the order in which you define them.
%% ---------------------------------------------------------------------
%% \nomacronym[sort]{label}{SHORT}{long form}
\newcommand{\nomacronym}[4][]{%
    \ifx\relax#1\relax
        \newabbreviation[type=acronyms, category=acronym, sort={#3}]{#2}{#3}{#4}%
    \else
        \newabbreviation[type=acronyms, category=acronym, sort={#1}]{#2}{#3}{#4}%
    \fi}
%% \nomdimless[sort]{label}{symbol}{description}{definition}       (symbol and definition: maths)
\newcommand{\nomdimless}[5][]{\thesis@nomentry{dimless}{#1}{#2}{#3}{#4}{definition={\thesis@definition{#5}}}}
%% \nomlatin[sort]{label}{symbol}{description}{dimensions}{units}   (symbol and dimensions: maths)
\newcommand{\nomlatin}[6][]{\thesis@nomentry{latin}{#1}{#2}{#3}{#4}{dimensions={\thesis@dims{#5}}, units={#6}}}
%% \nomgreek[sort]{label}{symbol}{description}{dimensions}{units}
\newcommand{\nomgreek}[6][]{\thesis@nomentry{greek}{#1}{#2}{#3}{#4}{dimensions={\thesis@dims{#5}}, units={#6}}}
%% \nomsuper[sort]{label}{symbol}{description}
\newcommand{\nomsuper}[4][]{\thesis@nomentry{super}{#1}{#2}{#3}{#4}{}}
%% \nomsub[sort]{label}{symbol}{description}
\newcommand{\nomsub}[4][]{\thesis@nomentry{sub}{#1}{#2}{#3}{#4}{}}

%% #1 type, #2 sort key (may be empty: then the label), #3 label, #4 symbol, #5 description, #6 extra keys
\newcommand{\thesis@nomentry}[6]{%
    \ifx\relax#2\relax
        \newglossaryentry{#3}{type=#1, sort={#3}, name={\ensuremath{#4}}, description={#5}, #6}%
    \else
        \newglossaryentry{#3}{type=#1, sort={#2}, name={\ensuremath{#4}}, description={#5}, #6}%
    \fi}
%% Definitions are set in full-size (display-style) maths, with room above and below so fractions
%% do not touch the next row.
\newcommand{\thesis@definition}[1]{\ensuremath{\displaystyle #1}\rule[-2.6ex]{0pt}{5.4ex}}
%% Dimensions are given in maths ("\Lunit\Tunit^{-1}"); "--" gives an en dash for dimensionless quantities.
\newcommand{\thesis@dims}[1]{%
    \def\thesis@arg{#1}\def\thesis@dash{--}%
    \ifx\thesis@arg\thesis@dash \textendash \else \ensuremath{#1}\fi}

%% MLT dimensions (https://www.ldeo.columbia.edu/~martins/hydro/case_studies/units_dimensions.html)
\newcommand{\Lunit}{\mathrm{L}}   % length
\newcommand{\Munit}{\mathrm{M}}   % mass
\newcommand{\Tunit}{\mathrm{T}}   % time
\newcommand{\Qunit}{\mathrm{Q}}   % temperature

%% ---------------------------------------------------------------------
%% Layout of the lists: same columns as before.
%% ---------------------------------------------------------------------
%% Narrow symbol, dimensions and units columns and 4 pt column gaps leave the description as much
%% room as possible, also on the narrow Dover page.
\newlength\symbolwidth      \setlength\symbolwidth{4em}
\newlength\dimensionswidth  \setlength\dimensionswidth{4.75em}
\newlength\unitswidth       \setlength\unitswidth{5.5em}
\newlength\descriptionwidth
\newlength\definitionwidth

\newcommand{\nomheader}[1]{{\headingfont\mbox{#1}}} % \mbox: no hyphenation in narrow columns
\newcommand{\nomtitle}[1]{\par\bigskip\noindent{\headingfont\LARGE #1}\par\nopagebreak\smallskip}

%% The lists get their titles from \printnom, so the glossary's own heading is switched off.
\renewcommand*{\glossarysection}[2][]{}

%% Common settings for the styles below.
\newcommand{\thesis@nomtablesetup}{%
    \setlength\tabcolsep{4pt}%
    \setlength\LTleft{0pt}\setlength\LTright{0pt}\setlength\LTpre{0pt}\setlength\LTpost{0pt}%
    \renewcommand*{\glossaryheader}{}%
    \renewcommand*{\glsgroupheading}[1]{}%
    \renewcommand*{\glsgroupskip}{}%
    \renewcommand*{\subglossentry}[3]{\glossentry{##2}{##3}}}

%% Symbol | Description          (acronyms: description starts with a capital)
\newglossarystyle{nom-plain}{%
    \thesis@nomtablesetup
    \renewenvironment{theglossary}{%
        \setlength\descriptionwidth{\dimexpr\textwidth-\symbolwidth-2\tabcolsep\relax}%
        \begin{longtable}{@{}p{\symbolwidth}p{\descriptionwidth}@{}}
            \nomheader{Symbol} & \nomheader{Description} \\[2pt] \endhead}
        {\end{longtable}}%
    \renewcommand*{\glossentry}[2]{%
        \glstarget{##1}{\glossentryname{##1}} & \RaggedRight\glossentrydesc{##1} \tabularnewline}}
\newglossarystyle{nom-acronyms}{%
    \setglossarystyle{nom-plain}%
    \renewcommand*{\glossentry}[2]{%
        \glstarget{##1}{\glossentryname{##1}} & \RaggedRight\Glsxtrlong{##1} \tabularnewline}}

%% Symbol | Description | Definition
\newglossarystyle{nom-definition}{%
    \thesis@nomtablesetup
    \renewenvironment{theglossary}{%
        \setlength\definitionwidth{\dimexpr\dimensionswidth+\unitswidth+2\tabcolsep\relax}%
        \setlength\descriptionwidth{\dimexpr\textwidth-\symbolwidth-\definitionwidth-4\tabcolsep\relax}%
        \begin{longtable}{@{}p{\symbolwidth}p{\descriptionwidth}p{\definitionwidth}@{}}
            \nomheader{Symbol} & \nomheader{Description} & \nomheader{Definition} \\[2pt] \endhead}
        {\end{longtable}}%
    \renewcommand*{\glossentry}[2]{%
        \glstarget{##1}{\glossentryname{##1}} & \RaggedRight\glossentrydesc{##1} & \glsentrydefinition{##1} \tabularnewline}}

%% Symbol | Description | Dimensions | Units
\newglossarystyle{nom-dimensions}{%
    \thesis@nomtablesetup
    \renewenvironment{theglossary}{%
        \setlength\descriptionwidth{\dimexpr\textwidth-\symbolwidth-\dimensionswidth-\unitswidth-6\tabcolsep\relax}%
        \begin{longtable}{@{}p{\symbolwidth}p{\descriptionwidth}p{\dimensionswidth}p{\unitswidth}@{}}
            \nomheader{Symbol} & \nomheader{Description} & \nomheader{Dimensions} & \nomheader{Units} \\[2pt] \endhead}
        {\end{longtable}}%
    \renewcommand*{\glossentry}[2]{%
        \glstarget{##1}{\glossentryname{##1}} & \RaggedRight\glossentrydesc{##1} & \glsentrydimensions{##1} & \glsentryunits{##1} \tabularnewline}}

%% \printnom{type}{title}{style}{sort}: prints one list with its title.
\newcommand{\printnom}[4]{%
    \nomtitle{#2}%
    \printnoidxglossary[type=#1, style=#3, sort=#4]}
\makeatother
