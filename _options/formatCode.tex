%% Code listings (listings package).
%% Digital edition: coloured syntax highlighting. Print editions: black and white (bold keywords,
%% grey italic comments), because printers often charge every page with colour as a colour page.
%% Colours with enough contrast against white for small text (4.5:1 or more): green 5.6, grey 5.7,
%% purple 6.6, blue 8.6.
\definecolor{codegreen}{rgb}{0,0.478,0}
\definecolor{codegrey}{gray}{0.4}
\definecolor{codepurple}{rgb}{0.58,0,0.82}
\lstset{
    frame=tb,
    language=Python,
    aboveskip=3mm,
    belowskip=3mm,
    showstringspaces=false,
    columns=flexible,
    basicstyle={\small\ttfamily},
    numbers=left,
    numbersep=5pt,
    numberstyle=\tiny\color{codegrey},
    breaklines=true,
    breakatwhitespace=false,
    tabsize=3,
    extendedchars=true,
    literate={á}{{\'a}}1 {ã}{{\~a}}1 {é}{{\'e}}1 {ü}{{\"u}}1,
}
\ifprintversion
    \lstset{
        keywordstyle=\bfseries,
        commentstyle=\itshape\color{codegrey},
        stringstyle=\color{black},
    }
\else
    \lstset{
        keywordstyle=\color{blue},
        commentstyle=\color{codegreen},
        stringstyle=\color{codepurple},
    }
\fi
