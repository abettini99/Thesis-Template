
\setcounter{secnumdepth}{2} % number chapters, sections and subsections; subsubsections and below are unnumbered
\renewcommand{\thechapter}{\arabic{chapter}}
%% Section numbers: Dutch/German convention with a trailing dot ("2.1.") in headings, the contents
%% and running heads, but not in cross-references, so \cref gives "Section 2.1" rather than "Section 2.1.".
\renewcommand{\thesection}{\thechapter.\arabic{section}}
\renewcommand{\thesubsection}{\thesection.\arabic{subsection}}
\renewcommand{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}
\setsecnumformat{\csname the#1\endcsname.\quad}
\renewcommand{\cftsectionaftersnum}{.}
\renewcommand{\cftsubsectionaftersnum}{.}

\def\blfootnote{\xdef\@thefnmark{}\@footnotetext}

%% Same heading styles for book (print) and report.

    %% Chapter formatting
    \if@print
        \openright % Makes sure that chapter is on recto page (clearing the verso page)
    \else
        \openany % report: chapters start on the next page, no blank pages
    \fi
    \makechapterstyle{mainmatter}{%
        % \renewcommand{\chapterheadstart}{}
        \renewcommand{\chapternamenum}{\space}
        % \renewcommand{\printchapternum}{}
        % \renewcommand{\afterchapternum}{}
        \renewcommand{\chapnamefont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\chapnumfont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\chaptitlefont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\printchaptername}{%
            \raggedleft \chapnamefont \@chapapp}
        \renewcommand{\printchaptertitle}[1]{%
            \raggedleft \chaptitlefont ##1}
        % \renewcommand{\afterchaptertitle}{%
        %     \vskip\onelineskip}
        \setlength{\beforechapskip}{50pt}
        \setlength{\midchapskip}{20pt}
        \setlength{\afterchapskip}{60pt}

        \renewcommand{\prechapterprecis}{%
            \vspace*{\prechapterprecisshift}%
            \precisfont\begin{Spacing}{1.10}} %% NOTE: Spacing in here mismatches with maintext! On purpose!
        \renewcommand{\postchapterprecis}{\end{Spacing}\clearpage}
        \renewcommand*{\precisfont}{}
    }
    
    \makechapterstyle{frontmatter}{%
        % \renewcommand{\chapterheadstart}{}
        \renewcommand{\chapternamenum}{\space}
        % \renewcommand{\printchapternum}{}
        % \renewcommand{\afterchapternum}{}
        \renewcommand{\chapnamefont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\chapnumfont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\chaptitlefont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\printchaptername}{%
            \raggedleft \chapnamefont \@chapapp}
        \renewcommand{\printchaptertitle}[1]{%
            \raggedleft \chaptitlefont ##1}
        % \renewcommand{\afterchaptertitle}{%
        %     \vskip\onelineskip}
        \setlength{\beforechapskip}{0pt}
        \setlength{\midchapskip}{20pt}
        \setlength{\afterchapskip}{20pt}
    }

    \makechapterstyle{bibliography}{%
        % \renewcommand{\chapterheadstart}{}
        \renewcommand{\chapternamenum}{\space}
        % \renewcommand{\printchapternum}{}
        % \renewcommand{\afterchapternum}{}
        \renewcommand{\chapnamefont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\chapnumfont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\chaptitlefont}{\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont}
        \renewcommand{\printchaptername}{%
            \raggedleft \chapnamefont \@chapapp}
        \renewcommand{\printchaptertitle}[1]{%
            \raggedleft \chaptitlefont ##1}
        % \renewcommand{\afterchaptertitle}{%
        %     \vskip\onelineskip}
        \setlength{\beforechapskip}{0pt}
        \setlength{\midchapskip}{20pt}
        \setlength{\afterchapskip}{20pt}
    }
    
    %% Section formatting
    \setbeforesecskip{1\onelineskip}
    \setsecindent{0em}
    \setsecheadstyle{\bfseries\headingfont}
    \setaftersecskip{0.1\onelineskip}

    %% Subsection formatting
    \setbeforesubsecskip{1\onelineskip}
    \setsubsecindent{0em}
    \setsubsecheadstyle{\bfseries\headingfont}
    \setaftersubsecskip{0.1\onelineskip}

    %% Subsubsection formatting
    \setbeforesubsubsecskip{1\onelineskip}
    \setsubsubsecindent{0em}
    \setsubsubsecheadstyle{\bfseries\headingfont}
    \setaftersubsubsecskip{0.1\onelineskip}

    %% paragraph formatting
    \setbeforeparaskip{1\onelineskip}
    \setparaindent{0em}
    \setparaheadstyle{\bfseries\headingfont}
    \setafterparaskip{-1.5em} % +values give vertical spacing, -values give horizontal spacing
    
    %% subparagraph formatting
    \setbeforesubparaskip{1\onelineskip}
    \setsubparaindent{1.5em}
    \setsubparaheadstyle{\bfseries\headingfont}
    \setaftersubparaskip{-1.5em} % +values give vertical spacing, -values give horizontal spacing
    
    %% Epigraph formatting
    \epigraphtextposition{flushright}
    \setlength{\epigraphwidth}{0.6\linewidth}
    \setlength{\epigraphrule}{0pt}
    \setlength{\beforeepigraphskip}{1\baselineskip}
    \setlength{\afterepigraphskip}{1\baselineskip}

    %% ToC, LoF, LoT formatting
    \renewcommand{\tocheadstart}{\chapterheadstart}
    \renewcommand{\lofheadstart}{\chapterheadstart}
    \renewcommand{\lotheadstart}{\chapterheadstart}
    \renewcommand{\printtoctitle}[1]{\hfill\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont Contents}
    \renewcommand{\printloftitle}[1]{\hfill\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont List of Figures}
    \renewcommand{\printlottitle}[1]{\hfill\fontsize{\chaptertitlesize}{\chaptertitlesize}\headingfont List of Tables}
    \renewcommand{\aftertoctitle}{\afterchaptertitle}
    \renewcommand{\afterloftitle}{\afterchaptertitle\insertchapterspace} % for some reason the spacing between the title and first entry of ToC and LoF/LoT is not the same.
    \renewcommand{\afterlottitle}{\afterchaptertitle\insertchapterspace}
