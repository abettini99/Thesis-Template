%% Maths font. MathTime Professional 2 when its OpenType file is in the Fonts folder (detected by the
%% class; see Fonts/README.md), otherwise newtx, the Times-style maths that ships with TeX Live.
\ifthesismathtime
%% ---------------------------------------------------------------------------------------------
%% MathTime Professional 2 (Full or Lite), through unicode-math. Lite has no bold italic, no bold
%% upright Greek, no calligraphic, blackboard bold or Fraktur. Its calligraphic letters and bold Latin
%% letters (vectors, tensors) then come from newtx, as without MathTime; bold Greek, blackboard bold
%% and Fraktur from TeX Gyre Termes Math, the Times-style OpenType maths font in TeX Live (unicode-math
%% treats Greek letters as Unicode characters, which the classic newtx fonts cannot supply).
%% ---------------------------------------------------------------------------------------------
%% unicode-math announces that it takes over a few symbols from mathtools (\overbracket, \coloneqq):
%% intended here, so it goes to the log as information instead of a warning.
\ExplSyntaxOn
\msg_redirect_name:nnn { unicode-math } { mathtools-overbracket } { info }
\msg_redirect_name:nnn { unicode-math } { mathtools-colon } { info }
\ExplSyntaxOff
\RequirePackage{unicode-math}
\setmathfont{MTPro2Math\thesismathtimeedition.otf}[Path=Fonts/, Scale=\thesis@fontscale]
\def\thesis@tmp{Lite}\ifx\thesismathtimeedition\thesis@tmp
\ifdefined\directlua
    \setmathfont{texgyretermes-math.otf}[range={bb,bbit,bfcal,frak,bffrak,bfit,bfsfit,bfup/{greek,Greek}},
        Scale=\thesis@fontscale]
    %% newtx calligraphic and bold italic, as classic maths alphabets (Latin letters), scaled like the text
    \edef\ntxmath@scaled{s*[\thesis@fontscale]}
    \DeclareFontEncoding{LMS}{}{}\DeclareFontSubstitution{LMS}{ntxsy}{m}{n} % newtx's symbol encoding
    \DeclareMathAlphabet{\thesis@ntxcal}{LMS}{ntxsy}{m}{n}
    \DeclareMathAlphabet{\thesis@ntxbfit}{OML}{ntxmi}{b}{it}
    \AddToHook{begindocument/end}{\let\mathcal\thesis@ntxcal}
    %% bold: a single Latin letter from newtx, anything else (Greek, digits) through \symbfit
    \ExplSyntaxOn
    \cs_new_protected:Npn \thesisboldmath #1
      {
        \bool_lazy_and:nnTF { \tl_if_single_token_p:n {#1} } { \token_if_letter_p:N #1 }
          { \thesis@ntxbfit {#1} } { \symbfit {#1} }
      }
    \ExplSyntaxOff
\else
    %% XeLaTeX cannot mix classic maths alphabets with unicode-math: everything Lite lacks comes from
    %% TeX Gyre Termes Math (the template is meant for LuaLaTeX).
    \setmathfont{texgyretermes-math.otf}[range={bb,bbit,cal,bfcal,frak,bffrak,bfit,bfsfit,bfup/{greek,Greek}},
        Scale=\thesis@fontscale]
    \newcommand{\thesisboldmath}{\symbfit}
\fi
\else
    \newcommand{\thesisboldmath}{\symbfit}

%% \mathsf (matrices \mathsf{A}, operators \mathsf{Res}) in Source Sans, the document's sans-serif font.

%% unicode-math redefines these at the start of the document; restore the template's meaning.
\AddToHook{begindocument/end}{%

    \renewcommand{\Re}{\mathrm{Re}}%   Reynolds number, not the real part
    \let\coloneqq\coloneq\let\eqqcolon\eqcolon}% := and =: as single glyphs

\else

%% newtx
%% ---------------------------------------------------------------------------------------------
\newcommand{\thesisboldmath}{\bm} % bold for \vec and \tensor (bm is loaded in packages.tex)
%% Math: newtxmath, the Times-style math font that matches the TeX Gyre Termes body text.
%% Variables are italic, as is standard for Times-based documents.
%% newtxmath takes digits, \mathrm, \mathbf, \mathit and operator names (sin, log, ...) from the current
%% roman font. Under fontspec that font has no math encoding, so it would fall back to Computer Modern.
%% Temporarily point it at newtx's own Times while loading, then restore the fontspec main font.
%% Scaled like the text (95 % for the Dover size): maths fonts via newtxmath's option, digits and
%% operator names via newtx's text-font scaling.
\edef\ntx@scaled{s*[\thesis@fontscale]}
\let\thesis@rmdefault\rmdefault
\renewcommand{\rmdefault}{ntxtlf}
\edef\thesis@tmp{\noexpand\RequirePackage[scaled=\thesis@fontscale]{newtxmath}}\thesis@tmp
\let\rmdefault\thesis@rmdefault

%% \mathsf (e.g. matrices \mathsf{A}, operators \mathsf{Res}) in Source Sans, the document's sans-serif font.
%% newtxmath declares it in the old OT1 encoding, for which fontspec fonts have no shapes, so TeX
%% substituted a default font (warning "Font shape OT1/SourceSans3/m/n undefined").
\DeclareMathAlphabet{\mathsf}{TU}{\sfdefault}{m}{n}
\SetMathAlphabet{\mathsf}{bold}{TU}{\sfdefault}{\bfdefault}{n}
\fi
