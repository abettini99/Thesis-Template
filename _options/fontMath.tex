%% Maths font, as chosen by the class (see personal.cls and Fonts/README.md): MathTime Professional 2
%% when tools/mtpro2_otf.py has written its OpenType fonts into the Fonts folder, otherwise newtx, the
%% Times-style maths that ships with TeX Live.
\ifthesismathtime
%% ---------------------------------------------------------------------------------------------
%% MathTime Professional 2 as OpenType maths fonts (mtpro2-otf, with unicode-math): placed exactly as
%% the classic Type 1 fonts are, every glyph maps to Unicode, and all of mtpro2's macros work (\xl..
%% \XXXL operators, \SQRT, \widearc, \bm, \hm, ...). Complete or Lite, whichever the files were made
%% from; Lite takes the AMS symbols, blackboard bold, Euler script and Fraktur, and newtx bold for \bm,
%% as the Type 1 set-up does. The package's options (Fonts/README.md) go in the list below.
%%   ntxcal: \mathcal from newtx (the simpler V and W)
%%   textsf: \mathsf (matrices \mathsf{A}, operators \mathsf{Res}) in Source Sans, the document's sans
%%   scale:  the maths at the size of the text (95 % for the Dover size)
%% (without unicode-math's notes that mathtools' colons and brackets stay mathtools' own: mtpro2-otf
%% keeps them, as with the Type 1 fonts)
\ExplSyntaxOn
\msg_redirect_name:nnn { unicode-math } { mathtools-colon } { none }
\msg_redirect_name:nnn { unicode-math } { mathtools-overbracket } { none }
\ExplSyntaxOff
\edef\thesis@tmp{\noexpand\RequirePackage[ntxcal, textsf, scale=\thesis@fontscale]{mtpro2-otf}}\thesis@tmp
%% (mtpro2-otf keeps the composite symbols of the Type 1 set-up itself: mathtools' colons and brackets,
%% amsmath's \iiiint and slanted \varTheta; the units' °, µ, Å and Ω are in its fonts)
%% (mtpro2-otf also turns maths in headings into its characters for the PDF bookmarks)
\newcommand{\thesisboldmath}{\bm} % mtpro2-otf's \bm (Lite: newtx bold letters)
\edef\thesismathtimeedition{\MTPOTFedition}
\ClassInfo{personal}{Maths font: MathTime Professional 2 \MTPOTFedition\space (OpenType, Fonts/)}
\else
%% ---------------------------------------------------------------------------------------------
%% newtx
%% ---------------------------------------------------------------------------------------------
\newcommand{\thesisboldmath}{\bm} % bold for \vec and \tensor (bm is loaded in packages.tex)
%% Math: newtxmath, the Times-style math font that matches the TeX Gyre Termes body text.
%% Variables are italic, as is standard for Times-based documents.
%% newtxmath takes digits, \mathrm, \mathbf, \mathit and operator names (sin, log, ...) from the current
%% roman font. Under fontspec that font has no math encoding, so it would fall back to Computer Modern.
%% Temporarily point it at newtx's own Times while loading, then restore the fontspec main font.
%% Scaled like the text (95 % for the Dover size): maths fonts via newtxmath's option, digits and
%% operator names via newtx's text-font scaling.
\edef\ntx@scaled{s*[\thesis@fontscale]}
\let\thesis@rmdefault\rmdefault
\renewcommand{\rmdefault}{ntxtlf}
\edef\thesis@tmp{\noexpand\RequirePackage[scaled=\thesis@fontscale]{newtxmath}}\thesis@tmp
\let\rmdefault\thesis@rmdefault

%% \mathsf (e.g. matrices \mathsf{A}, operators \mathsf{Res}) in Source Sans, the document's sans-serif font.
%% newtxmath declares it in the old OT1 encoding, for which fontspec fonts have no shapes, so TeX
%% substituted a default font (warning "Font shape OT1/SourceSans3/m/n undefined").
\DeclareMathAlphabet{\mathsf}{TU}{\sfdefault}{m}{n}
\SetMathAlphabet{\mathsf}{bold}{TU}{\sfdefault}{\bfdefault}{n}

%% Tagged PDF: the MathML that goes with each formula needs a Unicode maths font (the MathTime fonts),
%% and luamml stops the build with newtx. Here the formulas are tagged without MathML.
\ExplSyntaxOn
\tag_if_active:T { \tagpdfsetup { math/mathml/luamml/load = false } }
\ExplSyntaxOff
\fi

%% In PDF bookmarks (maths in a heading), the alphabets and bold commands just pass their letters on
%% (bm's \bm would stop the build there).
\AtBeginDocument{\ifdefined\pdfstringdefDisableCommands\pdfstringdefDisableCommands{%
    \@for\thesis@tmp:={mathrm,mathit,mathbf,mathsf,mathtt,mathcal,mathscr,mathfrak,mathbb,mathcurly,%
        symup,symit,symbf,symsf,symtt,symcal,symscr,symfrak,symbb,bm,hm,boldsymbol}\do{%
        \expandafter\let\csname\thesis@tmp\endcsname\@firstofone}}\fi}
