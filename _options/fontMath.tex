%% Maths font, as chosen by the class (see personal.cls and Fonts/README.md): MathTime Professional 2
%% when tools/mtpro2_unicode.py has prepared it in the Fonts folder, otherwise newtx, the Times-style
%% maths that ships with TeX Live.
\ifthesismathtime
%% ---------------------------------------------------------------------------------------------
%% MathTime Professional 2, classic Type 1 fonts through mtpro2.sty (LuaLaTeX), as prepared by
%% tools/mtpro2_unicode.py: every glyph carries a Unicode name, so maths copies correctly and the PDF/A
%% check passes. All of mtpro2's macros work: \xl..\XXXL operators, \SQRT, \widearc, \bm, \hm, ...
%% ---------------------------------------------------------------------------------------------
\input{mtpro2-glyphtounicode}        % Unicode for the renamed glyphs (variants, sizes)
\pdfextension mapfile{+mtpro2.map}  % which .pfb file belongs to which font
%% The Dover size sets the text at 95 %: scale the MathTime fonts with it (mtpro2.sty has no option).
\ifdim\thesis@fontscale pt<1pt
\directlua{
    local scale = \thesis@fontscale
    local define = luatexbase.remove_from_callback("define_font", "luaotfload.define_font")
    luatexbase.add_to_callback("define_font", function(name, size, id)
        if size > 0 and name:match("^mt2") then size = math.floor(size * scale + 0.5) end
        return define(name, size, id)
    end, "thesis.mtpro2-scale")
}
\fi
%% Complete: MathTime's own blackboard bold, script, Fraktur and AMS symbols. Lite has none of these: AMS
%% symbols and blackboard bold, Euler script and Fraktur (all in TeX Live) instead.
\def\thesis@tmp{Complete}\ifx\thesismathtimeedition\thesis@tmp
    \RequirePackage[complete, mtpbb, mtpscr, mtpfrak, nofontinfo]{mtpro2}
    \newcommand{\thesisboldmath}{\bm} % real bold (bm is loaded in packages.tex)
\else
    \RequirePackage{amssymb} % the AMS symbols Lite lacks (\varnothing, \nleq, ...), before mtpro2
    \RequirePackage[lite, amsbb, eufrak, nofontinfo]{mtpro2}
    \DeclareMathAlphabet{\mathscr}{U}{eus}{m}{n} % script: Euler (Lite has none)
    %% Lite has no bold maths: bold Latin letters (vectors, tensors) from newtx, scaled like the text
    \edef\ntxmath@scaled{s*[\thesis@fontscale]}
    \DeclareMathAlphabet{\thesis@ntxbfit}{OML}{ntxmi}{b}{it}
    \ExplSyntaxOn
    \cs_new_protected:Npn \thesisboldmath #1
      {
        \bool_lazy_and:nnTF { \tl_if_single_token_p:n {#1} } { \token_if_letter_p:N #1 }
          { \thesis@ntxbfit {#1} } { \bm {#1} }
      }
    \ExplSyntaxOff
\fi
%% Calligraphic letters from newtx (the simpler V and W), scaled like the text.
\edef\ntxmath@scaled{s*[\thesis@fontscale]}
\DeclareFontEncoding{LMS}{}{}\DeclareFontSubstitution{LMS}{ntxsy}{m}{n} % newtx's symbol encoding
\DeclareMathAlphabet{\thesis@ntxcal}{LMS}{ntxsy}{m}{n}
\AddToHook{begindocument/end}{\let\mathcal\thesis@ntxcal}
%% \mathsf (matrices \mathsf{A}, operators \mathsf{Res}) in Source Sans, the document's sans-serif font.
\DeclareMathAlphabet{\mathsf}{TU}{\sfdefault}{m}{n}
\SetMathAlphabet{\mathsf}{bold}{TU}{\sfdefault}{\bfdefault}{n}
\ClassInfo{personal}{Maths font: MathTime Professional 2 \thesismathtimeedition\space (Fonts/)}
\else
%% ---------------------------------------------------------------------------------------------
%% newtx
%% ---------------------------------------------------------------------------------------------
\newcommand{\thesisboldmath}{\bm} % bold for \vec and \tensor (bm is loaded in packages.tex)
%% Math: newtxmath, the Times-style math font that matches the TeX Gyre Termes body text.
%% Variables are italic, as is standard for Times-based documents.
%% newtxmath takes digits, \mathrm, \mathbf, \mathit and operator names (sin, log, ...) from the current
%% roman font. Under fontspec that font has no math encoding, so it would fall back to Computer Modern.
%% Temporarily point it at newtx's own Times while loading, then restore the fontspec main font.
%% Scaled like the text (95 % for the Dover size): maths fonts via newtxmath's option, digits and
%% operator names via newtx's text-font scaling.
\edef\ntx@scaled{s*[\thesis@fontscale]}
\let\thesis@rmdefault\rmdefault
\renewcommand{\rmdefault}{ntxtlf}
\edef\thesis@tmp{\noexpand\RequirePackage[scaled=\thesis@fontscale]{newtxmath}}\thesis@tmp
\let\rmdefault\thesis@rmdefault

%% \mathsf (e.g. matrices \mathsf{A}, operators \mathsf{Res}) in Source Sans, the document's sans-serif font.
%% newtxmath declares it in the old OT1 encoding, for which fontspec fonts have no shapes, so TeX
%% substituted a default font (warning "Font shape OT1/SourceSans3/m/n undefined").
\DeclareMathAlphabet{\mathsf}{TU}{\sfdefault}{m}{n}
\SetMathAlphabet{\mathsf}{bold}{TU}{\sfdefault}{\bfdefault}{n}
\fi
