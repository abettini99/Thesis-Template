%% Maths in a heading also goes into the PDF bookmarks. hyperref cannot read unicode-math's symbol commands
%% there (\alpha, \nabla, \int, ...: an error per symbol): in bookmarks, let each stand for its character.
\directlua{
    local chars, aliases
    function thesis_bookmark_symbols()
        local bs, pc, lb, rb = string.char(92), string.char(37), string.char(123), string.char(125)
        if not chars then
            chars = {}
            local f = io.open(kpse.find_file("unicode-math-table.tex"))
            local pattern = "UnicodeMathSymbol" .. lb .. '"(' .. pc .. "x+)" .. rb .. lb .. "." .. "(" .. pc .. "a+)"
                .. pc .. "s*" .. rb .. lb .. "." .. "(" .. pc .. "a+)"
            for cp, name, class in f:read("a"):gmatch(pattern) do
                cp = tonumber(cp, 16)
                if cp > 127 and not class:find("accent") and not class:find("radical")
                    and not class:find("over") and not class:find("under") then
                    chars[name] = chars[name] or utf8.char(cp)
                end
            end
            f:close()
            aliases = {le = "leq", ge = "geq", ne = "neq", to = "rightarrow", gets = "leftarrow"}
            for a, b in pairs(aliases) do chars[a] = chars[b] end % (unicode-math's other names)
        end
        local ct = luatexbase.registernumber("catcodetable@atletter")
        for name, ch in pairs(chars) do              % (not where a macro of the document's own stands)
            if aliases[name] or not token.create(name).cmdname:find("call") then
                tex.sprint(ct, bs .. "def" .. bs .. name .. "{" .. ch .. "}")
            end
        end
    end
}
