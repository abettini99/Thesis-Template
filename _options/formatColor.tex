%% TU Delft house style colours.
%% All colours are RGB or grey, which matches the sRGB colour profile declared for PDF/A
%% (see \DocumentMetadata in main.tex). Black, white and anthracite are pure grey, so black
%% text and thumb tabs print with black ink only.
%% Cyan uses the official RGB value from the TU Delft corporate design; the other colours are
%% colour-managed conversions of the template's original CMYK values (shown in the comments).
\definecolor{tudelft-cyan}{HTML}{00A6D6}         % CMYK 100/0/0/0 (official RGB)
\definecolor{tudelft-black}{gray}{0}              % CMYK 0/0/0/100
\definecolor{tudelft-white}{gray}{1}              % CMYK 0/0/0/0
\definecolor{tudelft-sea-green}{HTML}{71C9BB}    % CMYK 54/0/32/0
\definecolor{tudelft-green}{HTML}{0098A2}        % CMYK 100/15/40/0
\definecolor{tudelft-dark-blue}{HTML}{003C77}    % CMYK 100/66/0/40
\definecolor{tudelft-purple}{HTML}{22186A}       % CMYK 98/100/0/35
\definecolor{tudelft-turquoise}{HTML}{00A9BE}    % CMYK 82/0/21/8
\definecolor{tudelft-sky-blue}{HTML}{7AC7DD}     % CMYK 45/0/6/6
\definecolor{tudelft-lavendel}{HTML}{80A7D1}     % CMYK 45/20/0/7
\definecolor{tudelft-orange}{HTML}{F1893F}       % CMYK 2/56/84/0
\definecolor{tudelft-warm-purple}{HTML}{82278D}  % CMYK 58/100/0/2
\definecolor{tudelft-fuchsia}{HTML}{A80576}      % CMYK 19/100/0/19
\definecolor{tudelft-bright-green}{HTML}{B0D235} % CMYK 36/0/100/0
\definecolor{tudelft-yellow}{HTML}{FBF491}       % CMYK 2/0/54/0
