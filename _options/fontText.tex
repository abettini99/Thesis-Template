%% fontspec leaves the maths fonts alone (no-math): newtx or MathTime set them (fontMath.tex).
\RequirePackage[no-math]{fontspec}
%% Main text: TeX Gyre Termes, a free Times New Roman clone that ships with TeX Live (works on Overleaf).
%% Loaded by file name so it works with both XeLaTeX and LuaLaTeX.
\setmainfont{texgyretermes}[
    Extension = {.otf},
    UprightFont = {*-regular},
    BoldFont = {*-bold},
    ItalicFont = {*-italic},
    BoldItalicFont = {*-bolditalic},
    Ligatures = TeX,
    Scale = \thesis@fontscale   % 0.95 for the Dover size (9.5 pt), 1 otherwise
]

%% Heading font: Source Sans (Adobe, SIL Open Font License), included in TeX Live, so no font files
%% are needed in the project. Light for titles, semibold for section headings.
%% TeX Live 2026 renamed the fonts from Source Sans Pro to Source Sans 3 (file names SourceSans3-...);
%% use whichever is installed, so the template works with old and new TeX Live (and on Overleaf).
\IfFontExistsTF{SourceSans3-Light.otf}{\def\thesis@sourcesans{SourceSans3}}{\def\thesis@sourcesans{SourceSansPro}}
\newfontfamily{\sourcesans}{\thesis@sourcesans}[
    Extension   = {.otf},
    UprightFont = {*-Light},
    BoldFont    = {*-Semibold},
    Ligatures   = TeX,    % -- and --- become en and em dashes, as in the body text
    Scale       = \thesis@fontscale
]

%% Also the document's sans-serif font (\sffamily, \textsf), so nothing falls back to Latin Modern Sans.
\setsansfont{\thesis@sourcesans}[Extension={.otf}, UprightFont={*-Regular}, BoldFont={*-Semibold}, ItalicFont={*-RegularIt}, BoldItalicFont={*-SemiboldIt}, Ligatures=TeX, Scale=\thesis@fontscale]

%% Typewriter font (code, file names, URLs, DOIs): Source Code Pro (Adobe, in TeX Live), the monospaced
%% companion of Source Sans. Its weight sits well next to Times (Latin Mono looked thin); scaled to the
%% x-height of the text. Older TeX Live names the italic SourceCodePro-It, newer SourceCodePro-RegularIt.
\IfFontExistsTF{SourceCodePro-RegularIt.otf}{\def\thesis@tmp{RegularIt}}{\def\thesis@tmp{It}}
\setmonofont{SourceCodePro}[Extension=.otf, UprightFont=*-Regular, ItalicFont=*-\thesis@tmp,
    BoldFont=*-Semibold, BoldItalicFont=*-SemiboldIt, Scale=MatchLowercase]

%% Running heads: Source Sans Regular capitals, letterspaced (capitals need extra space between them
%% to read well), a size below the text. Page numbers in the same size, not spaced.
\newfontfamily{\runningheadcaps}{\thesis@sourcesans}[
    Extension   = {.otf},
    UprightFont = {*-Regular},
    LetterSpace = 8,
    Ligatures   = TeX,
    Scale       = \thesis@fontscale
]
\newcommand{\runningheadtext}[1]{{\small\runningheadcaps\MakeUppercase{#1}}}
\newcommand{\runningheadfolio}{\small\sffamily}

%% Heading font: used for chapter/section titles, running heads, page numbers and nomenclature headers.
%% To change the heading font everywhere, define another family above and change this one line.
\newcommand{\headingfont}{\sourcesans}
