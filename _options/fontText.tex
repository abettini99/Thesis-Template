\RequirePackage[no-math]{fontspec} % no-math: leave math fonts to newtxmath (see fontMath.tex)
%% Main text: TeX Gyre Termes, a free Times New Roman clone that ships with TeX Live (works on Overleaf).
%% Loaded by file name so it works with both XeLaTeX and LuaLaTeX.
\setmainfont{texgyretermes}[
    Extension = {.otf},
    UprightFont = {*-regular},
    BoldFont = {*-bold},
    ItalicFont = {*-italic},
    BoldItalicFont = {*-bolditalic},
    Ligatures = TeX
]

%% Very nice alternative to Futura Medium, which is not free. Found under Google Fonts
\newfontfamily{\didactgothic}{DidactGothic}[
    Extension = {.ttf},
    UprightFont = {*-Regular},
    Path = {./Fonts/DidactGothic/}
]

%% Quite close to Didact Gothic, with bold support. Ideally would like Didact Gothic but bold. Found under Google Fonts
\newfontfamily{\opensans}{OpenSans}[
    Extension = {.ttf},
    UprightFont = {*-Light},
    BoldFont = {*-Bold},
    Path = {./Fonts/OpenSans/}
]

%% Tahoma is not free, but Wine Tahoma is licensed under GNU Lesser General Public License (see Wikipedia link for Tahoma)
\newfontfamily{\winetahoma}{WineTahoma}[
    Extension = {.ttf},
    UprightFont = {*-Regular},
    BoldFont = {*-Bold},
    Path = {./Fonts/WineTahoma/}
]

%% Heading font: used for chapter/section titles, running heads, page numbers and nomenclature headers.
%% To change the heading font everywhere, change this one line (e.g. \winetahoma or \didactgothic).
\newcommand{\headingfont}{\opensans}
