\RequirePackage[no-math]{fontspec} % no-math: leave math fonts to newtxmath (see fontMath.tex)
%% Main text: TeX Gyre Termes, a free Times New Roman clone that ships with TeX Live (works on Overleaf).
%% Loaded by file name so it works with both XeLaTeX and LuaLaTeX.
\setmainfont{texgyretermes}[
    Extension = {.otf},
    UprightFont = {*-regular},
    BoldFont = {*-bold},
    ItalicFont = {*-italic},
    BoldItalicFont = {*-bolditalic},
    Ligatures = TeX
]
% \RequirePackage{newpxtext} %% previous main text (Palatino clone)

%% Heading font: Source Sans Pro (Adobe, SIL Open Font License), included in TeX Live, so no font
%% files are needed in the project. Light for titles, semibold for section headings.
\newfontfamily{\sourcesans}{SourceSansPro}[
    Extension   = {.otf},
    UprightFont = {*-Light},
    BoldFont    = {*-Semibold},
    Ligatures   = TeX     % -- and --- become en and em dashes, as in the body text
]

%% Also the document's sans-serif font (\sffamily, \textsf), so nothing falls back to Latin Modern Sans.
\setsansfont{SourceSansPro}[Extension={.otf}, UprightFont={*-Regular}, BoldFont={*-Semibold}, ItalicFont={*-RegularIt}, BoldItalicFont={*-SemiboldIt}, Ligatures=TeX]

%% Heading font: used for chapter/section titles, running heads, page numbers and nomenclature headers.
%% To change the heading font everywhere, define another family above and change this one line.
\newcommand{\headingfont}{\sourcesans}
