%% Page styles (memoir). Running heads in the heading font, uppercase.
%%   mainmatter     : chapter title on left pages, section on right pages, thumb tabs
%%   chapteronly    : chapter title on both pages (chapters without sections, the references)
%%   appendix       : "Appendix A - title" on both pages, thumb tabs
%%   chapteropening : chapter opening pages: only the thumb tab
%%   frontmatter    : page number centred at the bottom
\makepagestyle{mainmatter}
\makeevenhead{mainmatter}{\headingfont\lthumb\thepage}{\headingfont\MakeUppercase{\leftmark}}{}
\makeoddhead{mainmatter}{}{\headingfont\MakeUppercase{\rightmark}}{\headingfont\thepage\rthumb}
\makeevenfoot{mainmatter}{}{}{}
\makeoddfoot{mainmatter}{}{}{}
\makeheadrule{mainmatter}{\textwidth}{\normalrulethickness}

\makepagestyle{appendix}
\makeevenhead{appendix}{\headingfont\lthumb\thepage}{\headingfont\MakeUppercase{Appendix \thechapter~\textendash~\leftmark}}{}
\makeoddhead{appendix}{}{\headingfont\MakeUppercase{Appendix \thechapter~\textendash~\leftmark}}{\headingfont\thepage\rthumb}
\makeevenfoot{appendix}{}{}{}
\makeoddfoot{appendix}{}{}{}
\makeheadrule{appendix}{\textwidth}{\normalrulethickness}

%% Chapter opening pages: no running head, but the thumb tab (the page a reader is looking for).
\makepagestyle{chapteropening}
\makeevenhead{chapteropening}{\lthumb}{}{}
\makeoddhead{chapteropening}{}{}{\rthumb}
\makeevenfoot{chapteropening}{}{}{}
\makeoddfoot{chapteropening}{}{}{}

\makepagestyle{frontmatter}
\makeevenhead{frontmatter}{}{}{}
\makeoddhead{frontmatter}{}{}{}
\makeevenfoot{frontmatter}{}{\headingfont\thepage}{}
\makeoddfoot{frontmatter}{}{\headingfont\thepage}{}

%% plain: used for chapter opening pages in the front and back matter (page number at the bottom)
\makeevenfoot{plain}{}{\headingfont\thepage}{}
\makeoddfoot{plain}{}{\headingfont\thepage}{}

\makepagestyle{chapteronly}
\makeevenhead{chapteronly}{\headingfont\lthumb\thepage}{\headingfont\MakeUppercase{\leftmark}}{}
\makeoddhead{chapteronly}{}{\headingfont\MakeUppercase{\leftmark}}{\headingfont\thepage\rthumb}
\makeevenfoot{chapteronly}{}{}{}
\makeoddfoot{chapteronly}{}{}{}
\makeheadrule{chapteronly}{\textwidth}{\normalrulethickness}

\renewcommand{\chaptermark}[1]{%
\markboth{#1}{}} 
%% The section number is stored together with its title, so the running head never pairs
%% the title of one section with the number of another.
\renewcommand{\sectionmark}[1]{%
\markright{\thesection.\hspace{1em}#1}}
