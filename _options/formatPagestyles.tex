%% Page styles (memoir). Running heads in letterspaced capitals, page numbers in the same size
%% (\runningheadtext and \runningheadfolio in fontText.tex). New page styles have empty heads and feet.
%%   mainmatter     : chapter title on left pages, section on right pages, thumb tabs
%%   chapteronly    : chapter title on both pages (chapters without sections, the references)
%%   appendix       : "Appendix A - title" on both pages, thumb tabs. With "Appendix A - " in front, a long
%%                    title runs into the page number: give it a short one for the head,
%%                    \chapter[title in the contents][title in the head]{title}
%%   chapteropening : chapter opening pages: only the thumb tab
%%   frontmatter    : page number centred at the bottom
\makepagestyle{mainmatter}
\makeevenhead{mainmatter}{\runningheadfolio\lthumb\thepage}{\runningheadtext{\leftmark}}{}
\makeoddhead{mainmatter}{}{\runningheadtext{\rightmark}}{\runningheadfolio\thepage\rthumb}
\makeheadrule{mainmatter}{\textwidth}{0.3pt}

\makepagestyle{appendix}
\makeevenhead{appendix}{\runningheadfolio\lthumb\thepage}{\runningheadtext{Appendix \thechapter~\textendash~\leftmark}}{}
\makeoddhead{appendix}{}{\runningheadtext{Appendix \thechapter~\textendash~\leftmark}}{\runningheadfolio\thepage\rthumb}
\makeheadrule{appendix}{\textwidth}{0.3pt}

%% Chapter opening pages: no running head, but the thumb tab (the page a reader is looking for).
\makepagestyle{chapteropening}
\makeevenhead{chapteropening}{\lthumb}{}{}
\makeoddhead{chapteropening}{}{}{\rthumb}

\makepagestyle{frontmatter}
\makeevenhead{frontmatter}{}{}{}
\makeoddhead{frontmatter}{}{}{}
\makeevenfoot{frontmatter}{}{\runningheadfolio\thepage}{}
\makeoddfoot{frontmatter}{}{\runningheadfolio\thepage}{}

%% plain: page number at the bottom. Chapter opening pages have no page number anywhere in the book
%% (main.tex: empty in the front and back matter and for the references, chapteropening in the chapters).
\makeevenfoot{plain}{}{\runningheadfolio\thepage}{}
\makeoddfoot{plain}{}{\runningheadfolio\thepage}{}

\makepagestyle{chapteronly}
\makeevenhead{chapteronly}{\runningheadfolio\lthumb\thepage}{\runningheadtext{\leftmark}}{}
\makeoddhead{chapteronly}{}{\runningheadtext{\leftmark}}{\runningheadfolio\thepage\rthumb}
\makeheadrule{chapteronly}{\textwidth}{0.3pt}

\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
%% The section number is stored together with its title, so the running head never pairs
%% the title of one section with the number of another.
\renewcommand{\sectionmark}[1]{\markright{\thesection.\enspace#1}}
